```src/build.sh
# Menu paket dan arsitektur. Semua tampilan ke STDERR; hanya pilihan ke STDOUT.
# Nomor paket bersifat GLOBAL lintas halaman, jadi bisa langsung diketik.

menu_pkg(){
    local per=18 cols=3
    local all=() d
    [ -d "$TPL" ] || die "folder template tidak ada: $TPL"
    for d in "$TPL"/*/; do
        [ -f "$d/template" ] || continue
        all+=("$(basename "$d")")
    done
    local total=${#all[@]}
    [ "$total" -gt 0 ] || die "tidak ada template di $TPL"

    local pages=$(( (total + per - 1) / per )) page=0
    local start count rows r c li gi colw maxlen cell ans tok sel out

    while :; do
        start=$(( page * per ))
        count=$(( total - start ))
        [ "$count" -le "$per" ] || count=$per
        rows=$(( (count + cols - 1) / cols ))

        maxlen=0
        for (( li=0; li<count; li++ )); do
            [ ${#all[$((start+li))]} -le "$maxlen" ] || maxlen=${#all[$((start+li))]}
        done
        colw=$(( maxlen + 8 ))

        printf '[+] Template: %d total — halaman %d/%d\n' \
            "$total" "$((page+1))" "$pages" >&2
        for (( r=0; r<rows; r++ )); do
            for (( c=0; c<cols; c++ )); do
                li=$(( c * rows + r ))
                [ "$li" -lt "$count" ] || continue
                gi=$(( start + li ))
                cell="[$((gi+1))] ${all[$gi]}"
                printf '%-*s' "$colw" "$cell" >&2
            done
            printf '\n' >&2
        done

        read -rp "[?] Nomor (pisah spasi) / nama paket / n p q: " ans
        case "$ans" in
            q|Q) die "dibatalkan" ;;
            n|N) page=$(( (page + 1) % pages )); continue ;;
            p|P) page=$(( (page - 1 + pages) % pages )); continue ;;
            "")  die "tidak ada package dipilih" ;;
        esac

        out=""
        for tok in $ans; do
            if [[ $tok =~ ^[0-9]+$ ]]; then
                # 10#: paksa desimal ("08"/"09" bukan oktal); panjang dibatasi cegah overflow
                if [ ${#tok} -le 6 ] && [ "$((10#$tok))" -ge 1 ] && [ "$((10#$tok))" -le "$total" ]; then
                    sel="${all[$((10#$tok - 1))]}"
                else
                    warn "nomor $tok di luar rentang 1..$total"; continue
                fi
            elif pkg_safe "$tok" && [ -f "$TPL/$tok/template" ]; then
                sel="$tok"
            else
                warn "tidak ada template: $tok"; continue
            fi
            in_list "$out" "$sel" || out="$out $sel"
        done
        if [ -n "$out" ]; then printf '%s\n' $out; return 0; fi
        warn "tidak ada pilihan valid, ulangi"
    done
}

menu_arch(){
    local pkg="$1" sup=() i a n def ch out=""
    mapfile -t sup < <(tpl_supported "$pkg")
    [ ${#sup[@]} -gt 0 ] || die "$pkg: tidak ada arsitektur yang didukung"

    echo "[+] Architecture — $pkg" >&2
    i=1
    for a in "${ALL[@]}"; do
        if in_list "${sup[*]}" "$a"; then printf '    [%d] %s\n' "$i" "$a" >&2; fi
        i=$((i+1))
    done
    printf '    [%d] all\n' "$(( ${#ALL[@]} + 1 ))" >&2

    if in_list "${sup[*]}" "$HOST_ARCH"; then def="$HOST_ARCH"; else def="${sup[0]}"; fi
    read -rp "[?] Choose (pisah spasi, contoh: 1 3 6) [default: $def]: " ch
    if [ -z "$ch" ]; then printf '%s\n' "$def"; return 0; fi

    local num
    for i in $ch; do
        num=""
        if [[ "$i" =~ ^[0-9]{1,6}$ ]]; then num=$((10#$i)); fi    # 10#: "08"/"09" bukan oktal
        if { [ -n "$num" ] && [ "$num" -eq "$(( ${#ALL[@]} + 1 ))" ]; } || [ "$i" = all ]; then
            printf '%s\n' "${sup[@]}"; return 0
        fi
        if [ -n "$num" ] && [ "$num" -ge 1 ] && [ "$num" -le ${#ALL[@]} ]; then
            n="${ALL[$((num-1))]}"
            if in_list "${sup[*]}" "$n"; then
                in_list "$out" "$n" || out="$out $n"
            else
                warn "[$i] $n tidak didukung $pkg"
            fi
        elif in_list "${sup[*]}" "$i"; then
            in_list "$out" "$i" || out="$out $i"
        else
            warn "pilihan '$i' diabaikan"
        fi
    done
    [ -n "$out" ] || die "tidak ada arsitektur valid dipilih"
    printf '%s\n' $out
}

build_summary(){
    local pkg="$1" okl="$2" badl="$3" nok nbad
    nok=$(printf '%s' "$okl" | wc -w)
    nbad=$(printf '%s' "$badl" | wc -w)
    printf '\n' >&2
    printf '=== %s ===\n' "$pkg" >&2
    printf "${G}[+]${N} Success : %d Archs\n" "$nok" >&2
    printf "${G}[+]${N} %s\n" "${okl:-None}" >&2
    printf "${R}[-]${N} Failed : %d Archs\n" "$nbad" >&2
    printf "${R}[-]${N} %s\n" "${badl:-None}" >&2
}

cmd_build(){
    local pkgs=() req=() r pkg t sup want targets x f
    local okl="" badl="" noarch=0 rc=0

    while [ $# -gt 0 ]; do
        case "$1" in
            -a|--arch) shift; while [ $# -gt 0 ] && [[ $1 != -* ]]; do req+=("$1"); shift; done ;;
            *) pkgs+=("$1"); shift ;;
        esac
    done

    if [ ${#pkgs[@]} -eq 0 ]; then mapfile -t pkgs < <(menu_pkg); fi
    [ ${#pkgs[@]} -gt 0 ] || die "tidak ada package"

    # Validasi SEMUA nama + template dulu, SEBELUM clone/bootstrap yang berat.
    for pkg in "${pkgs[@]}"; do
        pkg_safe "$pkg" || die "nama paket tidak aman (harus [A-Za-z0-9._+-], tidak diawali '.' atau '-'): '$pkg'"
        [ -f "$TPL/$pkg/template" ] || die "template tidak ada: $TPL/$pkg/template"
    done

    cmd_setup

    for pkg in "${pkgs[@]}"; do
        mapfile -t sup < <(tpl_supported "$pkg")
        [ ${#sup[@]} -gt 0 ] || { warn "$pkg: archs kosong setelah ekspansi, skip"; continue; }

        want=()
        if [ ${#req[@]} -gt 0 ]; then
            for r in "${req[@]}"; do
                mapfile -t -O ${#want[@]} want < <(expand_archs "$r")
            done
        else
            mapfile -t want < <(menu_arch "$pkg")
        fi

        targets=()
        for x in "${want[@]}"; do
            in_list "${sup[*]}" "$x" || continue
            in_list "${targets[*]}" "$x" || targets+=("$x")
        done
        [ ${#targets[@]} -gt 0 ] || { warn "$pkg: tidak ada irisan arch, skip"; continue; }

        info "$pkg support : ${sup[*]}"
        info "$pkg dibuild : ${targets[*]}"
        tpl_stage "$pkg"

        okl=""; badl=""; noarch=0
        if tpl_is_noarch "$pkg"; then
            noarch=1
            info "$pkg noarch — satu build untuk ${#targets[@]} tag"
        fi

        for x in "${targets[@]}"; do
            if [ "$noarch" = 1 ] && [ "$x" != "${targets[0]}" ]; then continue; fi
            f=()
            mapfile -t f < <(flags_for "$x")
            info "Building: xbps-src ${f[*]:-} pkg $pkg  ($x)"
            if (cd "$VP" && ./xbps-src ${f[@]+"${f[@]}"} pkg "$pkg"); then
                ok "$pkg $x → $VP/hostdir/binpkgs/"
                if [ "$noarch" = 1 ]; then okl="${targets[*]}"; else okl="$okl $x"; fi
            else
                err "gagal: $pkg $x — lanjut ke arsitektur berikutnya"
                if [ "$noarch" = 1 ]; then badl="${targets[*]}"; else badl="$badl $x"; fi
                rc=1
            fi
        done

        build_summary "$pkg" "${okl# }" "${badl# }"
    done

    return $rc
}
```

```src/common.sh
# Konstanta & helper bersama. TIDAK ada side effect selain assignment.
# Semua log ke STDERR — stdout dipakai untuk data (dipanggil via mapfile).
REPO_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
VP="$REPO_DIR/void-packages"
TPL="$REPO_DIR/pkg-template"
GH_REPO="T4n-Labs/repo-t4nos"
ALL=(x86_64 x86_64-musl aarch64 aarch64-musl armv6l armv6l-musl armv7l armv7l-musl i686)
HOST_ARCH="$(xbps-uhelper arch 2>/dev/null || uname -m)"

B='\033[0;34m'; G='\033[0;32m'; Y='\033[1;33m'; R='\033[0;31m'; N='\033[0m'
info(){ printf "${B}[INFO]${N} %s\n" "$*" >&2; }
ok(){   printf "${G}[OK]${N} %s\n"   "$*" >&2; }
warn(){ printf "${Y}[WARN]${N} %s\n" "$*" >&2; }
err(){  printf "${R}[ERROR]${N} %s\n" "$*" >&2; }
die(){  err "$*"; exit 1; }

need(){ command -v "$1" >/dev/null 2>&1 || die "$1 belum terinstal"; }

# charset nama asset yang aman di GitHub Releases (terverifikasi: '+' aman, '~' tidak)
name_safe(){ printf '%s' "$1" | grep -qE '^[A-Za-z0-9._+-]+$'; }

# Nama paket untuk operasi path (rm/cp): allowlist name_safe + tidak boleh diawali
# '.' atau '-' (menolak '.', '..', '.hidden', opsi). Regex name_safe sengaja tidak diubah.
pkg_safe(){ name_safe "$1" || return 1; case "$1" in .*|-*) return 1 ;; esac; return 0; }

in_list(){ case " $1 " in *" $2 "*) return 0 ;; esac; return 1; }
```

```src/env.sh
# Environment build: clone, bootstrap, sync template, flag xbps-src.

cmd_setup(){
    # 1. Clone atau update void-packages
    if [ -d "$VP/.git" ]; then
        ok "void-packages sudah ada"
        if [ -f "$VP/.git/shallow" ]; then
            info "Clone shallow terdeteksi — mengambil history penuh (bisa lama)..."
            (cd "$VP" && git fetch --unshallow) || warn "unshallow gagal, lanjut dengan yang ada"
        fi
    else
        info "Clone void-packages (full history, bisa >1 GB dan lama)..."
        git clone https://github.com/void-linux/void-packages "$VP"
    fi

    # 2. Bootstrap semua 9 arsitektur
    local arch ok_count=0 fail_count=0 skip_count=0
    for arch in "${ALL[@]}"; do
        local mdir="$VP/masterdir-$arch"
        if [ -d "$mdir" ]; then
            info "masterdir-$arch sudah ada — dilewati"
            skip_count=$((skip_count+1))
            continue
        fi
        info "binary-bootstrap untuk $arch ..."
        if (cd "$VP" && ./xbps-src -A "$arch" binary-bootstrap); then
            ok "masterdir-$arch siap"
            ok_count=$((ok_count+1))
        else
            warn "masterdir-$arch gagal di-bootstrap"
            fail_count=$((fail_count+1))
        fi
    done

    # 3. Enable restricted packages
    if ! grep -q "^XBPS_ALLOW_RESTRICTED=yes" "$VP/etc/conf" 2>/dev/null; then
        mkdir -p "$VP/etc"
        echo "XBPS_ALLOW_RESTRICTED=yes" >> "$VP/etc/conf"
        ok "XBPS_ALLOW_RESTRICTED=yes diaktifkan di $VP/etc/conf"
    else
        ok "XBPS_ALLOW_RESTRICTED=yes sudah aktif"
    fi

    info "ringkasan setup: $ok_count dibuat, $skip_count sudah ada, $fail_count gagal"
    if [ "$fail_count" -gt 0 ]; then
        warn "beberapa arsitektur gagal di-bootstrap, tapi setup tetap lanjut"
    fi
}

cmd_sync(){
    [ -d "$VP/.git" ] || cmd_setup
    info "git pull void-packages..."
    (cd "$VP" && git pull --ff-only) || warn "pull gagal, lanjut dengan yang ada"
    _staged_prune
    local n=0 d name
    for d in "$TPL"/*/; do
        name=$(basename "$d"); [ -f "$d/template" ] || continue
        pkg_safe "$name" || { warn "sync: nama tidak aman, dilewati: $name"; continue; }
        if ! diff -rq "$d" "$VP/srcpkgs/$name" >/dev/null 2>&1; then
            rm -rf "${VP:?}/srcpkgs/$name"; cp -a "$d" "$VP/srcpkgs/$name"
            info "sync: $name"; n=$((n+1))
        fi
        _staged_add "$name"
    done
    ok "$n template disinkronkan"
}

cmd_update(){
    # Pastikan void-packages sudah di-setup
    if [ ! -d "$VP" ] || [ ! -x "$VP/xbps-src" ]; then
        err "update: void-packages belum di-setup. Jalankan: tools --setup"
        return 1
    fi

    info "Menjalankan xbps-src bootstrap-update di $VP ..."
    if (cd "$VP" && ./xbps-src bootstrap-update); then
        ok "bootstrap-update selesai"
    else
        warn "bootstrap-update gagal (exit $?), cek log di atas"
        return 1
    fi
}

# Daftar nama yang pernah kita salin ke $VP/srcpkgs (di dalam .git → tidak mengotori
# `git status`). Dipakai untuk membersihkan salinan template yang sudah dihapus dari
# pkg-template/. Pelacakan dimulai sejak sync/stage pertama dengan versi ini.
_staged_file(){ printf '%s\n' "$VP/.git/t4n-staged"; }
_staged_add(){
    local f; [ -d "$VP/.git" ] || return 0
    f=$(_staged_file)
    grep -qxF -- "$1" "$f" 2>/dev/null || printf '%s\n' "$1" >> "$f"
    return 0
}
_staged_prune(){
    local f keep name; f=$(_staged_file); [ -f "$f" ] || return 0
    keep=$(mktemp)
    while IFS= read -r name; do
        [ -n "$name" ] || continue
        if pkg_safe "$name" && [ ! -f "$TPL/$name/template" ]; then
            if git -C "$VP" ls-files --error-unmatch -- "srcpkgs/$name/template" >/dev/null 2>&1; then
                # ada di upstream Void: pulihkan resep resmi, buang sisa berkas kita
                git -C "$VP" checkout -- "srcpkgs/$name" \
                  && git -C "$VP" clean -fdq -- "srcpkgs/$name" \
                  && info "sync: $name tak ada lagi di pkg-template → dipulihkan dari void-packages" \
                  || warn "sync: gagal memulihkan srcpkgs/$name"
            else
                rm -rf "${VP:?}/srcpkgs/$name"
                info "sync: $name tak ada lagi di pkg-template → salinan dibersihkan"
            fi
        else
            printf '%s\n' "$name" >> "$keep"
        fi
    done < "$f"
    mv "$keep" "$f"
}

tpl_stage(){
    pkg_safe "$1" || die "tpl_stage: nama paket tidak aman: '$1'"
    [ -f "$TPL/$1/template" ] || die "tpl_stage: template tidak ada: $1"
    rm -rf "${VP:?}/srcpkgs/$1"; cp -a "$TPL/$1" "$VP/srcpkgs/$1"
    _staged_add "$1"
}

# flags_for <arsitektur-target>
#   -A = masterdir/host, -a = target cross (STRING PENUH, termasuk -musl)
flags_for(){
    local target="$1"
    in_list "${ALL[*]}" "$target" || die "flags_for: '$target' bukan arsitektur valid"

    local hb="${HOST_ARCH%-musl}" tb="${target%-musl}"
    local hl=glibc tl=glibc hm f=()
    if [[ $HOST_ARCH == *-musl ]]; then hl=musl; fi
    if [[ $target    == *-musl ]]; then tl=musl; fi
    # masterdir host dengan libc target. glibc TIDAK punya suffix.
    hm="$hb"
    if [ "$tl" = musl ]; then hm="${hb}-musl"; fi

    if [ "$target" = "$HOST_ARCH" ]; then
        :                                       # identik → native, tanpa flag
    elif [ "$tb" = "$hb" ]; then
        f+=(-A "$target")                       # base sama, libc beda
    elif [ "$tb" = i686 ] && [ "$hb" = x86_64 ] && [ "$hl" = "$tl" ]; then
        f+=(-A i686)                            # native 32-bit (cara resmi Void)
    else
        if [ "$hl" != "$tl" ]; then f+=(-A "$hm"); fi
        f+=(-a "$target")                       # PENUH: armv6l-musl, bukan armv6l
    fi

    if [ ${#f[@]} -gt 0 ]; then printf '%s\n' "${f[@]}"; fi
    return 0
}
```

```src/git.sh
# Tawaran push / PR. Aman tanpa TTY: EOF pada stdin = jawab "tidak".
# Selalu mengembalikan 0 — dipanggil setelah upload sukses, jangan menggagalkan exit code.
git_offer(){
    local ans="" origin
    read -rp "Git push perubahan sekarang? [y/N] " ans || ans=""
    [[ ${ans,,} == y* ]] || return 0
    git -C "$REPO_DIR" push || warn "push gagal"
    origin=$(git -C "$REPO_DIR" remote get-url origin 2>/dev/null || echo "")
    case "$origin" in
        *"$GH_REPO"*) ;;
        *) ans=""
           read -rp "Buka Pull Request ke $GH_REPO? [y/N] " ans || ans=""
           if [[ ${ans,,} == y* ]]; then
               gh pr create --repo "$GH_REPO" --fill || warn "gh pr create gagal"
           fi ;;
    esac
    return 0
}
```

```src/template.sh
# Parser template: awk/read, TIDAK pernah di-source (template dari pihak lain).
# Bisa di-source sendiri oleh CI / Let-X: . vdev/src/template.sh
#
# KONTRAK PARSER — harus identik dengan .github/workflows/gen-json.py:
#   field  : baris pertama yang cocok ^[ \t]*KEY= ; nilai = sisa baris.
#            '#' memulai komentar HANYA di luar kutip dan di awal nilai / setelah
#            spasi atau tab (seperti shell; "https://x/#a" dan "A # B" aman).
#            Backslash meloloskan karakter berikutnya (kecuali dalam kutip tunggal).
#            Lalu trim spasi/tab/CR, lalu buang SATU pasang kutip yang membungkus.
#   archs  : koma = spasi; kosong / "noarch" (saja) = semua tag; token "all" = "*";
#            token berawalan satu atau lebih "~" = negasi (jumlah "~" tidak
#            dibedakan: ~~i686 ≡ ~i686); pola glob; tanpa token positif = semua tag.

tpl_field(){
    [ -f "$1" ] || return 0
    awk -v key="$2" '
    BEGIN { sq = "\047"; dq = "\042" }
    function clean(s,   i, n, c, q, prev, out, f, l) {
        n = length(s); q = ""; prev = " "; out = ""; i = 1
        while (i <= n) {
            c = substr(s, i, 1)
            if (c == "\\" && q != sq && i < n) {
                out = out c substr(s, i + 1, 1); prev = substr(s, i + 1, 1); i += 2; continue
            }
            if (q != "") { if (c == q) q = ""; out = out c; prev = c; i++; continue }
            if (c == sq || c == dq) { q = c; out = out c; prev = c; i++; continue }
            if (c == "#" && (prev == " " || prev == "\t")) break
            out = out c; prev = c; i++
        }
        sub(/^[ \t\r]+/, "", out); sub(/[ \t\r]+$/, "", out)
        l = length(out)
        if (l >= 2) {
            f = substr(out, 1, 1)
            if ((f == sq || f == dq) && substr(out, l, 1) == f) out = substr(out, 2, l - 2)
        }
        return out
    }
    {
        if (match($0, "^[ \t]*" key "=")) { print clean(substr($0, RLENGTH + 1)); exit }
    }' "$1"
}

tpl_subpkgs(){
    local l w ws=()
    l=$(tpl_field "$1" subpackages)
    l="${l//\$\{pkgname\}/$2}"; l="${l//\$pkgname/$2}"
    read -ra ws <<< "$l" || true                 # tanpa glob terhadap isi cwd
    for w in ${ws[@]+"${ws[@]}"}; do printf '%s\n' "$w"; done
    return 0
}

# archs: kosong/noarch = 9; all = *; glob x86_64* = glibc+musl; ~token = negasi.
expand_archs(){
    local line="${1//,/ }" tok neg_t a keep toks=() pos=() neg=() out=()
    read -ra toks <<< "$line" || true            # read -a: tanpa glob terhadap cwd
    if [ ${#toks[@]} -eq 0 ] || { [ ${#toks[@]} -eq 1 ] && [ "${toks[0]}" = noarch ]; }; then
        printf '%s\n' "${ALL[@]}"; return 0
    fi
    for tok in "${toks[@]}"; do
        if [[ $tok == '~'* ]]; then
            neg_t="$tok"
            while [[ $neg_t == '~'* ]]; do neg_t="${neg_t#\~}"; done
            if [ "$neg_t" = all ]; then neg_t='*'; fi
            if [ -n "$neg_t" ]; then neg+=("$neg_t"); fi
        else
            if [ "$tok" = all ]; then tok='*'; fi
            pos+=("$tok")
        fi
    done
    for a in "${ALL[@]}"; do
        keep=0
        if [ ${#pos[@]} -eq 0 ]; then keep=1
        else for tok in "${pos[@]}"; do case "$a" in $tok) keep=1;; esac; done; fi
        if [ "$keep" = 1 ] && [ ${#neg[@]} -gt 0 ]; then
            for tok in "${neg[@]}"; do case "$a" in $tok) keep=0;; esac; done
        fi
        if [ "$keep" = 1 ]; then out+=("$a"); fi
    done
    if [ ${#out[@]} -gt 0 ]; then printf '%s\n' "${out[@]}"; fi
    return 0
}

tpl_supported(){ expand_archs "$(tpl_field "$TPL/$1/template" archs)"; }
tpl_is_noarch(){ [ "$(tpl_field "$TPL/$1/template" archs)" = noarch ]; }
```

```src/upload.sh
route_tags(){ # filename arch supported[] → tag tujuan
    local fn="$1" arch="$2" sup="$3" out=() x
    if [ "$arch" = noarch ]; then printf '%s\n' $sup; return 0; fi
    for x in $sup; do if [ "$x" = "$arch" ]; then out+=("$x"); fi; done
    if [ ${#out[@]} -gt 0 ]; then printf '%s\n' "${out[@]}"; fi
    return 0    # kosong = tidak ada tag; pemanggil yang memutuskan (bukan exit code)
}

cmd_upload(){
    need gh
    local pkgs=("$@")
    if [ ${#pkgs[@]} -eq 0 ]; then mapfile -t pkgs < <(menu_pkg); fi
    [ ${#pkgs[@]} -gt 0 ] || die "tidak ada package"

    # 1) belum login ≠ 2) login tapi tanpa izin write: dua kondisi, dua pesan.
    if ! gh auth status >/dev/null 2>&1; then
        err "gh belum terautentikasi — jalankan: gh auth login (upload & trigger dilewati)"
        git_offer; return 1
    fi
    local can_push
    if ! can_push=$(gh api "repos/$GH_REPO" --jq '.permissions.push' 2>&1); then
        err "gagal memeriksa izin ke $GH_REPO: $can_push"
        return 1
    fi
    if [ "$can_push" != true ]; then
        warn "Tidak punya akses write ke $GH_REPO — upload & trigger dilewati."
        info "Biner akan di-build dan di-upload maintainer setelah PR di-merge."
        git_offer; return
    fi

    local touched="" pkg t v rv names fn base arch sup n up x tags
    shopt -s nullglob
    for pkg in "${pkgs[@]}"; do
        pkg_safe "$pkg" || { warn "nama paket tidak aman, skip: $pkg"; continue; }
        t="$TPL/$pkg/template"; [ -f "$t" ] || { warn "template hilang: $pkg"; continue; }
        v=$(tpl_field "$t" version); rv=$(tpl_field "$t" revision); rv="${rv:-1}"
        names="$pkg
$(tpl_subpkgs "$t" "$pkg")"
        sup=$(tpl_supported "$pkg" | xargs)
        up=0
        for f in "$VP"/hostdir/binpkgs/*.xbps; do
            fn=$(basename "$f"); base="${fn%.xbps}"; arch="${base##*.}"
            name_safe "$fn" || { warn "nama asset tidak aman, skip: $fn"; continue; }
            n=$(xbps-uhelper getpkgname "${base%.*}") || continue
            printf '%s\n' "$names" | grep -qxF "$n" || continue
            case "$fn" in *"-${v}_${rv}."*) ;; *) warn "versi lain, skip: $fn"; continue ;; esac
            tags=$(route_tags "$fn" "$arch" "$sup") || tags=""
            [ -n "$tags" ] || { warn "$arch di luar archs: $fn"; continue; }
            for x in $tags; do
                gh release view "$x" -R "$GH_REPO" >/dev/null 2>&1 \
                  || gh release create "$x" -R "$GH_REPO" --prerelease --title "repository T4nOS for $x"
                gh release upload "$x" "$f" -R "$GH_REPO" --clobber
                ok "↑ $x/$fn"; up=$((up+1))
                in_list "$touched" "$x" || touched="$touched $x"
            done
        done
        [ "$up" -gt 0 ] || warn "$pkg: tidak ada biner cocok (build dulu?)"
    done
    if [ -n "$touched" ]; then
        info "Trigger workflow sekali untuk:$touched"
        gh workflow run process.yml -R "$GH_REPO" -f tags="$(echo $touched | xargs)"
        ok "Signing/index jalan di CI. Pantau: gh run watch -R $GH_REPO"
    fi
    git_offer
}
```
